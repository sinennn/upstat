/**
 * Gateway session store — the bearer credential the Go backend issues after
 * `UserService/GoogleAuth` and validates on every authenticated RPC
 * (`util.AuthenticateInterceptor`, HS256 JWT).
 *
 * This is deliberately NOT an auth system: it is the storage half of the
 * `AuthProvider` seam, shared by the provider that *mints* the session
 * (auth/gateway.ts) and the transport that *spends* it
 * (models/repositories/grpc/transport.ts). Keeping it in one place means the
 * token never has to travel through controllers or views.
 *
 * sessionStorage (not localStorage) — the credential dies with the tab/
 * webview, matching the fleet session standard and keeping the Tauri desktop
 * build from persisting a token to disk.
 */

const SESSION_KEY = "upstat.gateway-session";

export interface GatewaySession {
  token: string;
  user: AuthUser;
}

function storage(): Storage | null {
  if (typeof window === "undefined") return null;
  return window.sessionStorage;
}

export function readGatewaySession(): GatewaySession | null {
  const store = storage();
  if (!store) return null;
  const raw = store.getItem(SESSION_KEY);
  if (!raw) return null;
  try {
    const parsed = JSON.parse(raw) as GatewaySession;
    if (!parsed?.token || !parsed?.user) return null;
    return parsed;
  } catch {
    // a corrupt entry is treated as "signed out", never a crash
    return null;
  }
}

export function writeGatewaySession(session: GatewaySession): void {
  storage()?.setItem(SESSION_KEY, JSON.stringify(session));
}

export function clearGatewaySession(): void {
  storage()?.removeItem(SESSION_KEY);
}

/** Bearer token for repository calls; null when signed out. */
export function getSessionToken(): string | null {
  return readGatewaySession()?.token ?? null;
}

  }

      return "nodata";
    case "paused":
      return "paused";
    case "pending":
      return "pending";
  }
}

/** Monitors controller — uptime checks CRUD + mute (pages.md B7/B8). */
export function useMonitorsController() {
  const state = useRequest(() => monitorsRepo.list(), []);

  const create = useCallback(
    async (
      input: Pick<Monitor, "name" | "target" | "type"> & Partial<Monitor>,
    ) => {
      const monitor = await monitorsRepo.create(input);
      await state.reload();
      return monitor;
    },
    [state],
  );

  const setMuted = useCallback(
    async (id: string, muted: boolean) => {
      await monitorsRepo.update(id, { muted });
      await state.reload();
    },
    [state],
  );

  const remove = useCallback(
    async (id: string) => {
      await monitorsRepo.remove(id);
      await state.reload();
    },
    [state],
  );

  return { ...state, create, setMuted, remove };
}

/** B7 overview — UptimeCard strips for the first three live checks. */
export function useUptimeCardsController(monitors: Monitor[] | null) {
  const ids = (monitors ?? [])
    .filter((m) => m.status !== "pending")
    .slice(0, 3)
    .map((m) => m.id)
    .join(",");
  return useRequest(async () => {
    if (!ids) return [];
    return Promise.all(ids.split(",").map((id) => monitorsRepo.history(id)));
  }, [ids]);
}

import type * as grpcWeb from "grpc-web";

import { env } from "@/config/env";
import { getSessionToken } from "@/auth/session";
import { MonitorServiceClient } from "@/proto/MonitorServiceClientPb";
import { UserServiceClient } from "@/proto/UserServiceClientPb";
import { ApiError } from "../../errors";

/**
 * gRPC-Web transport — the real backend seam for the uptime/monitoring
 * vertical.
 *
 *   view → controller → repository (grpc/*) → this transport →
 *   gRPC-Web → Envoy (:8082) → api/common gRPC (:8080) → Mongo
 *
 * This module and its siblings under `grpc/` are the ONLY place the app
 * speaks gRPC; views never see it (enforced by `check:boundaries`) and
 * controllers never touch it.
 */

/**
 * gRPC status codes we can receive (grpc-web `StatusCode` enum values).
 * Declared numerically rather than imported from the enum so this file does
 * not need a value import of `grpc-web` (types only).
 */
const GRPC_INVALID_ARGUMENT = 3;
const GRPC_NOT_FOUND = 5;
const GRPC_ALREADY_EXISTS = 6;
const GRPC_PERMISSION_DENIED = 7;
const GRPC_FAILED_PRECONDITION = 9;
const GRPC_UNIMPLEMENTED = 12;
const GRPC_INTERNAL = 13;
const GRPC_UNAVAILABLE = 14;
const GRPC_UNAUTHENTICATED = 16;

/**
 * gRPC status → (HTTP-equivalent status, stable snake_case code).
 *
 * The repositories speak the ecosystem error envelope (models/errors.ts);
 * gRPC has its own code space, so every RPC failure is normalized here —
 * once, at the boundary — instead of leaking `RpcError` into controllers.
 */
function envelopeFor(code: number): { status: number; code: string } {
  switch (code) {
    case GRPC_INVALID_ARGUMENT:
      return { status: 422, code: "bad_shape" };
    case GRPC_NOT_FOUND:
      // Cross-tenant reads always read `not_found` (errors.ts contract):
      // the backend returns NotFound for another owner's monitor.
      return { status: 404, code: "not_found" };
    case GRPC_ALREADY_EXISTS:
      return { status: 409, code: "already_exists" };
    case GRPC_PERMISSION_DENIED:
      return { status: 403, code: "forbidden" };
    case GRPC_FAILED_PRECONDITION:
      return { status: 412, code: "failed_precondition" };
    case GRPC_UNIMPLEMENTED:
      return { status: 501, code: "unimplemented" };
    case GRPC_UNAUTHENTICATED:
      return { status: 401, code: "unauthenticated" };
    case GRPC_UNAVAILABLE:
      return { status: 503, code: "unavailable" };
    case GRPC_INTERNAL:
    default:
      return { status: 500, code: "internal" };
  }
}

let monitorClient: MonitorServiceClient | null = null;
let userClient: UserServiceClient | null = null;

/** `proto.MonitorService` client, pointed at the Envoy gRPC-Web endpoint. */
export function monitorClientFor(): MonitorServiceClient {
  if (!monitorClient) monitorClient = new MonitorServiceClient(env.envoyUrl);
  return monitorClient;
}

/** `proto.UserService` client, pointed at the Envoy gRPC-Web endpoint. */
export function userClientFor(): UserServiceClient {
  if (!userClient) userClient = new UserServiceClient(env.envoyUrl);
  return userClient;
}

/**
 * Request metadata carrying the gateway session. `authorization` is on the
 * Envoy CORS allow-list (deploy/helm/envoy/envoy.yaml), so the browser
 * preflight passes it through untouched.
 */
export function authMetadata(): grpcWeb.Metadata {
  const token = getSessionToken();
  return token ? { authorization: `Bearer ${token}` } : {};
}

export interface RpcCallOptions {
  /** Send the bearer session (authenticated RPCs). Default: true. */
  authenticated?: boolean;
}

/**
 * Runs a unary gRPC-Web call and normalizes failures into `ApiError`.
 *
 * `RpcError.code` is the numeric gRPC status and `message` is the backend's
 * `status.Error` text (e.g. "monitor not found"); both are preserved so the
 * UI copy stays truthful to what the service actually rejected.
 */
export async function call<T>(
  run: (metadata: grpcWeb.Metadata) => Promise<T>,
  options: RpcCallOptions = {},
): Promise<T> {
  const authenticated = options.authenticated ?? true;
  const metadata = authenticated ? authMetadata() : {};
  try {
    return await run(metadata);
  } catch (e) {
    const rpc = e as { code?: number; message?: string };
    if (typeof rpc?.code === "number") {
      const { status, code } = envelopeFor(rpc.code);
      throw new ApiError(status, {
        code,
        message: rpc.message || "the monitoring service rejected the request",
      });
    }
    // Transport-level failure (Envoy unreachable, CORS, offline): the
    // backend never answered, so report it as unavailable rather than
    // pretending it was a bad request.
    throw new ApiError(503, {
      code: "unavailable",
      message: e instanceof Error ? e.message : "the backend is unreachable",
    });
  }
}

/**
 * An RPC the current backend does not expose. Used by the gRPC repositories
 * for UI affordances whose contracts exist in the design but have no
 * service method yet — surfaces as a normal `ApiError` (501) so callers can
 * degrade honestly instead of silently succeeding.
 */
export function unimplemented(rpc: string): ApiError {
  return new ApiError(501, {
    code: "unimplemented",
    message: `${rpc} is not implemented by the monitoring backend yet`,
  });
}
